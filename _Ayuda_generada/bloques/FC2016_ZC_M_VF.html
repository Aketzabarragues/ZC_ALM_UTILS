    //  ===========================================================================================================

    //  en caso de que este abierto el mismo dispositivo.

                    "DB2016_M_VF".Mux[#for_mux_cambios].M_VF := "DB2016_M_VF".M_VF[#t_Index];

            END_IF;

        END_FOR;

    END_IF;
END_REGION

	    END_FOR;

	    //  ACUSE DE NUEVAS ALARMAS
	    IF #Ack THEN
	        FOR #for_i := 0 TO "N_MAX_DISP_AGRUP" DO
