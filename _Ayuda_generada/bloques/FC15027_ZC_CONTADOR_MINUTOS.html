Interfaz de Variables

SecciónNombreTipo de DatoDescripción
VAR_INPUT Pulso1seg Bool Pulso de 1 segundo
VAR_INPUT Iniciar Bool Orden iniciar contador

VAR MAX_MINUTOS UDInt := 10000000 Valor maximo de minutos

🔸 Lógica: RESET_CONTADOR

Este bloque se encarga de resetear el tiempo del contador

REGION RESET_CONTADOR
    IF #Reset OR #Contador.Minuto >= #MAX_MINUTOS THEN
        #Contador.Minuto := #Contador.Segundo := 0;
    END_IF;
END_REGION

🔸 Lógica: GESTION_CONTADOR

Este bloque se encarga de gestionar el contaje del contador de minutos, incrementando mediante la entrada #Pulso1seg

REGION GESTION_CONTADOR
    IF #Iniciar AND #Pulso1seg THEN
        #Contador.#Segundo += 1;
        IF #Contador.#Segundo >= 60 THEN
            #Contador.#Segundo := 0;
            #Contador.#Minuto += 1;
        END_IF;
    END_IF;
END_REGION

   END_VAR

   VAR_IN_OUT

   END_VAR

   VAR CONSTANT

   END_VAR

	        #Contador.Minuto := #Contador.Segundo := 0;
	    END_IF;
	    
	END_REGION
	
	
	//==========================================================================================================
	/// <RegionDoc>

	    IF #Iniciar AND #Pulso1seg THEN
	        #Contador.#Segundo += 1;
	        IF #Contador.#Segundo >= 60 THEN
	            #Contador.#Segundo := 0;
	            #Contador.#Minuto += 1;
	        END_IF;
	    END_IF;
	    
	END_REGION
	
	
END_FUNCTION
