REGION GESTION_CAMBIOS
    //  ===========================================================================================================
    //  Si existe cambios en el dispositivo generados por SCADA, actualizamos los datos en el resto de multiplexados,
    //  en caso de que este abierto el mismo dispositivo.
    IF "DB2025_TOT".Mux[#for_mux].ExisteCambio THEN

        //  Recorremos todos los multiplexados
        FOR #for_mux_cambios := 0 TO "N_MAX_MUX_DISP" DO

            //  No miramos el multiplexado actual
            IF #for_mux_cambios <> #for_mux THEN

                //  En caso de que este seleccionado el mismo indice, movemos el dispositivo al multiplexado
                //  correspondiente
                IF #t_Index = "DB2025_TOT".Mux[#for_mux_cambios].Index THEN
                    "DB2025_TOT".Mux[#for_mux_cambios].TOT := "DB2025_TOT".TOT[#t_Index];
                END_IF;

            END_IF;

        END_FOR;
        "DB2025_TOT".Mux[#for_mux].ExisteCambio := FALSE;
    END_IF;
END_REGION

REGION DIAGNOSIS

    //  RESET DEL TOTALIZADOR PARCIAL

    IF "DB2025_TOT".TOT[#for_i].Orden_ResetTotal OR "DB2025_TOT".TOT[#for_i].Estado_DesbordeTotal = #NUMERO_MAXIMO_DESBORDE THEN
        "DB2025_TOT".TOT[#for_i].Estado_Total := 0.0;
        "DB2025_TOT".TOT[#for_i].Estado_DesbordeTotal := 0;
        "DB2025_TOT".TOT[#for_i].Orden_ResetTotal := FALSE;
    END_IF;

    // =============================================================================
    //  RESET DEL TOTALIZADOR AUXILIAR (PARA UTILIZAR EN PROCESO)
    IF "DB2025_TOT".TOT[#for_i].Orden_ResetProceso THEN
        "DB2025_TOT".TOT[#for_i].Estado_Proceso := 0.0;
        "DB2025_TOT".TOT[#for_i].Orden_ResetProceso := FALSE;
    END_IF;

    // =============================================================================
    //  RESET DEL TOTALIZADOR TOTAL POR DESBORDAMIENTO
    IF "DB2025_TOT".TOT[#for_i].Estado_Total >= #VALOR_MAXIMO_CONTADORES THEN
        "DB2025_TOT".TOT[#for_i].Estado_DesbordeTotal += 1;
        "DB2025_TOT".TOT[#for_i].Estado_Total := 0.0;
    END_IF;

    // =============================================================================
    //  RESET DEL TOTALIZADOR PARCIAL POR DESBORDAMIENTO
    IF "DB2025_TOT".TOT[#for_i].Estado_Parcial >= #VALOR_MAXIMO_CONTADORES THEN
        "DB2025_TOT".TOT[#for_i].Estado_DesbordeParcial += 1;
        "DB2025_TOT".TOT[#for_i].Estado_Parcial := 0.0;
    END_IF;
END_REGION

	                    IF (("DB2025_TOT".Mux[#for_mux].TOT.Config_Habilitar <> "DB2025_TOT".TOT[#t_Index].Config_Habilitar) OR
	                        ("DB2025_TOT".Mux[#for_mux].TOT.Config_Tipo <> "DB2025_TOT".TOT[#t_Index].Config_Tipo) OR
	                        ("DB2025_TOT".Mux[#for_mux].TOT.Config_IncrDecr <> "DB2025_TOT".TOT[#t_Index].Config_IncrDecr) OR
	                        ("DB2025_TOT".Mux[#for_mux].TOT.Config_DireccionEntradaByte <> "DB2025_TOT".TOT[#t_Index].Config_DireccionEntradaByte) OR
	                        ("DB2025_TOT".Mux[#for_mux].TOT.Config_DireccionEntradaBit <> "DB2025_TOT".TOT[#t_Index].Config_DireccionEntradaBit) OR
	                        ("DB2025_TOT".Mux[#for_mux].TOT.Config_LitroXPulso <> "DB2025_TOT".TOT[#t_Index].Config_LitroXPulso)
	                        )
	                    THEN
	                        "DB2025_TOT".TOT[#t_Index].Config_Habilitar := "DB2025_TOT".Mux[#for_mux].TOT.Config_Habilitar;
	                        "DB2025_TOT".TOT[#t_Index].Config_Tipo := "DB2025_TOT".Mux[#for_mux].TOT.Config_Tipo;
	                        "DB2025_TOT".TOT[#t_Index].Config_IncrDecr := "DB2025_TOT".Mux[#for_mux].TOT.Config_IncrDecr;
	                        "DB2025_TOT".TOT[#t_Index].Config_DireccionEntradaByte := "DB2025_TOT".Mux[#for_mux].TOT.Config_DireccionEntradaByte;
	                        "DB2025_TOT".TOT[#t_Index].Config_DireccionEntradaBit := "DB2025_TOT".Mux[#for_mux].TOT.Config_DireccionEntradaBit;
	                        "DB2025_TOT".TOT[#t_Index].Config_LitroXPulso := "DB2025_TOT".Mux[#for_mux].TOT.Config_LitroXPulso;
	                        "DB2025_TOT".Mux[#for_mux].ExisteCambio := true;
	                    END_IF;
	                    
	                END_REGION
	                
	                
	                REGION GESTION_CAMBIOS
	                    
	                    //  ===========================================================================================================
	                    //  Si existe cambios en el dispositivo generados por SCADA, actualizamos los datos en el resto de multiplexados,
	                    //  en caso de que este abierto el mismo dispositivo.
	                    IF "DB2025_TOT".Mux[#for_mux].ExisteCambio THEN
	                        
	                        //  Recorremos todos los multiplexados
	                        FOR #for_mux_cambios := 0 TO "N_MAX_MUX_DISP" DO
	                            
	                            //  No miramos el multiplexado actual
	                            IF #for_mux_cambios <> #for_mux THEN
	                                
	                                //  En caso de que este seleccionado el mismo indice, movemos el dispositivo al multiplexado
	                                //  correspondiente
	                                IF #t_Index = "DB2025_TOT".Mux[#for_mux_cambios].Index THEN
	                                    "DB2025_TOT".Mux[#for_mux_cambios].TOT := "DB2025_TOT".TOT[#t_Index];
	                                END_IF;
	                                
	                            END_IF;
	                            
	                        END_FOR;
	                        "DB2025_TOT".Mux[#for_mux].ExisteCambio := FALSE;
	                    END_IF;
	                    
	                END_REGION
	                
	            ELSE
	                
	                //  ===========================================================================================================
	                //  En caso de que el indice sea 0, movemos el valor de la TOT[0] que no se usa para escribir el index.
	                "DB2025_TOT".Mux[#for_mux].TOT := "DB2025_TOT".TOT[0];
	                
	            END_IF;
	            
	        ELSE
	            
	            "DB2025_TOT".Mux[#for_mux].ExisteCambio := FALSE;
	            "DB2025_TOT".Mux[#for_mux].Index := 
	            "DB2025_TOT".Mux[#for_mux].oldIndex := 0;
	            
	        END_IF;
	        
	    END_FOR;
	    
	END_REGION
	
	
	REGION TIEMPO_CICLO_INTEGRACION
	    
	    
	    // =============================================================================
	    //  Tiempo ciclo para integración EA
	    "DB2025_TOT".TiempoCiclo.Actual := LREAL_TO_REAL(RUNTIME("DB2025_TOT".TiempoCiclo.auxMem));
	    "DB2025_TOT".TiempoCiclo.Actual := ("DB2025_TOT".TiempoCiclo.Actual * 1000.0);
	    
	    
	END_REGION
	
	
	REGION LOGICA_DEL_DISPOSITIVO
	    
	    
	    FOR #for_i := 1 TO "N_MAX_DISP_TOT" DO
	        
	        // =============================================================================
	        // DISPOSITIVO HABILITADO
	        IF "DB2025_TOT".TOT[#for_i].Config_Habilitar THEN
	            
	            
	            REGION MODO_TRABAJO
	                
	                IF "DB2025_TOT".TOT[#for_i].Orden_Activar THEN
	                    // =============================================================================
	                    // 0 = TOTALIZADOR POR PULSOS
	                    // 1 = TOTALIZADOR POR EA
	                    IF NOT "DB2025_TOT".TOT[#for_i].Config_Tipo THEN    //  Caudalímetro por Pulso
	                        
	                        REGION LECTURA_VALORES_DE_LA_TARJETA
	                            
	                            IF NOT #Simulacion THEN
	                                //  Caudalímetro por Pulsos - Contar pulsos
	                                IF "DB2025_TOT".TOT[#for_i].Config_DireccionEntradaByte > 0 THEN
	                                    
	                                    // =============================================================================
	                                    //  LECTURA ENTRADA DIGITAL
	                                    "DB2025_TOT".TOT[#for_i].Estado_Pulso := PEEK_BOOL(area := 16#81,
	                                                                                       dbNumber := 0,
	                                                                                       byteOffset := "DB2025_TOT".TOT[#for_i].Config_DireccionEntradaByte,
	                                                                                       bitOffset := "DB2025_TOT".TOT[#for_i].Config_DireccionEntradaBit);
	                                END_IF;
	                            ELSE
	                                "DB2025_TOT".TOT[#for_i].Estado_Pulso := #Pulso1seg;
	                            END_IF;
	                            
	                        END_REGION
	                        
	                        
	                        IF "DB2025_TOT".TOT[#for_i].Estado_Pulso AND NOT "DB2025_TOT".TOT[#for_i].Aux_oldActivado THEN
	                            
	                            IF NOT "DB2025_TOT".TOT[#for_i].Config_IncrDecr THEN
	                                "DB2025_TOT".TOT[#for_i].Estado_Parcial += "DB2025_TOT".TOT[#for_i].Config_LitroXPulso;
	                                "DB2025_TOT".TOT[#for_i].Estado_Total += "DB2025_TOT".TOT[#for_i].Config_LitroXPulso;
	                                "DB2025_TOT".TOT[#for_i].Estado_Proceso += "DB2025_TOT".TOT[#for_i].Config_LitroXPulso;
	                            ELSE
	                                "DB2025_TOT".TOT[#for_i].Estado_Parcial -= "DB2025_TOT".TOT[#for_i].Config_LitroXPulso;
	                                "DB2025_TOT".TOT[#for_i].Estado_Total -= "DB2025_TOT".TOT[#for_i].Config_LitroXPulso;
	                                "DB2025_TOT".TOT[#for_i].Estado_Proceso -= "DB2025_TOT".TOT[#for_i].Config_LitroXPulso;
	                            END_IF;
	                            
	                        END_IF;
	                        
	                        // =============================================================================
	                        // CALCULO TEORICO L/H
	                        IF #Pulso1seg THEN
	                            "DB2025_TOT".TOT[#for_i].Aux_AcuTiempo += 1;
	                        END_IF;
	                        
	                        IF "DB2025_TOT".TOT[#for_i].Aux_AcuTiempo > 2 THEN
	                            #t_LReal := "DB2025_TOT".TOT[#for_i].Estado_Total - "DB2025_TOT".TOT[#for_i].Aux_TotalizadorMem;
	                            
	                            "DB2025_TOT".TOT[#for_i].Estado_Valor := LREAL_TO_REAL(#t_LReal) * 3600.0 / "DB2025_TOT".TOT[#for_i].Aux_AcuTiempo;
	                            
	                            "DB2025_TOT".TOT[#for_i].Aux_AcuTiempo := 0;
	                            "DB2025_TOT".TOT[#for_i].Aux_TotalizadorMem := "DB2025_TOT".TOT[#for_i].Estado_Total;
	                        END_IF;
	                        
	                    ELSE
	                        //Totalizador por EA
	                        #t_LReal := (("DB2025_TOT".TOT[#for_i].Estado_Valor / 3600000.0 * "DB2025_TOT".TiempoCiclo.Actual));
	                        #t_LReal := #t_LReal + (#"Offset_l_h" / 3600000.0 * "DB2025_TOT".TiempoCiclo.Actual);
	                        
	                        IF NOT "DB2025_TOT".TOT[#for_i].Config_IncrDecr THEN
	                            "DB2025_TOT".TOT[#for_i].Estado_Parcial += #t_LReal;
	                            "DB2025_TOT".TOT[#for_i].Estado_Total += #t_LReal;
	                            "DB2025_TOT".TOT[#for_i].Estado_Proceso += #t_LReal;
	                        ELSE
	                            "DB2025_TOT".TOT[#for_i].Estado_Parcial -= #t_LReal;
	                            "DB2025_TOT".TOT[#for_i].Estado_Total -= #t_LReal;
	                            "DB2025_TOT".TOT[#for_i].Estado_Proceso -= #t_LReal;
	                        END_IF;
	                        
	                    END_IF;
	                    
	                END_IF;
	                
	                "DB2025_TOT".TOT[#for_i].Aux_oldActivado := "DB2025_TOT".TOT[#for_i].Estado_Pulso;
	                //Tiempo ciclo para integración EA
	                
	            END_REGION
	            
	            
	            REGION DIAGNOSIS
	                
	                // =============================================================================
	                //  RESET DEL TOTALIZADOR PARCIAL
	                IF "DB2025_TOT".TOT[#for_i].Orden_ResetParcial OR "DB2025_TOT".TOT[#for_i].Estado_DesbordeParcial = #NUMERO_MAXIMO_DESBORDE THEN
	                    "DB2025_TOT".TOT[#for_i].Estado_Parcial := 0.0;
	                    "DB2025_TOT".TOT[#for_i].Estado_DesbordeParcial := 0;
	                    "DB2025_TOT".TOT[#for_i].Orden_ResetParcial := FALSE;
	                END_IF;
	                
	                // =============================================================================
	                //  RESET DEL TOTALIZADOR TOTAL
	                IF "DB2025_TOT".TOT[#for_i].Orden_ResetTotal OR "DB2025_TOT".TOT[#for_i].Estado_DesbordeTotal = #NUMERO_MAXIMO_DESBORDE THEN
	                    "DB2025_TOT".TOT[#for_i].Estado_Total := 0.0;
	                    "DB2025_TOT".TOT[#for_i].Estado_DesbordeTotal := 0;
	                    "DB2025_TOT".TOT[#for_i].Orden_ResetTotal := FALSE;
	                END_IF;
	                
	                // =============================================================================
	                //  RESET DEL TOTALIZADOR AUXILIAR (PARA UTILIZAR EN PROCESO)
	                IF "DB2025_TOT".TOT[#for_i].Orden_ResetProceso THEN
	                    "DB2025_TOT".TOT[#for_i].Estado_Proceso := 0.0;
	                    "DB2025_TOT".TOT[#for_i].Orden_ResetProceso := FALSE;
	                END_IF;
	                
	                // =============================================================================
	                //  RESET DEL TOTALIZADOR TOTAL POR DESBORDAMIENTO
	                IF "DB2025_TOT".TOT[#for_i].Estado_Total >= #VALOR_MAXIMO_CONTADORES THEN
	                    "DB2025_TOT".TOT[#for_i].Estado_DesbordeTotal += 1;
	                    "DB2025_TOT".TOT[#for_i].Estado_Total := 0.0;
	                END_IF;
	                
	                // =============================================================================
	                //  RESET DEL TOTALIZADOR PARCIAL POR DESBORDAMIENTO
	                IF "DB2025_TOT".TOT[#for_i].Estado_Parcial >= #VALOR_MAXIMO_CONTADORES THEN
	                    "DB2025_TOT".TOT[#for_i].Estado_DesbordeParcial += 1;
	                    "DB2025_TOT".TOT[#for_i].Estado_Parcial := 0.0;
	                END_IF;
	                
	            END_REGION
	            
	        ELSE
	            
	            // =============================================================================
	            // DISPOSITIVO NO HABILITADO
	            // =============================================================================
	            "DB2025_TOT".TOT[#for_i].Estado_Valor := 0.0;
	            "DB2025_TOT".TOT[#for_i].Estado_Parcial := 0.0;
	            "DB2025_TOT".TOT[#for_i].Estado_Total := 0.0;
	            "DB2025_TOT".TOT[#for_i].Estado_Proceso := 0.0;
	            
	        END_IF;
	        
	    END_FOR;
	    
	    
	    
	END_REGION
	
	
	REGION TIEMPO_DE_CICLO
	    
	    #t_RT_Dummy := RUNTIME("DB2025_TOT".TiempoCiclo.auxMem);
	    
	END_REGION
	
	
END_FUNCTION
