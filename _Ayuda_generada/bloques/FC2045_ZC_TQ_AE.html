	                    //  =============================================================================
	                    //  GESTION TIEMPO FLIP FLOP
	                    IF #Pulso1seg THEN
	                        
	                        //  =============================================================================
	                        //  CUENTA FLIP
	                        IF "DB2045_TQ_AE".AE[#for_i].Aux_DuchaFlipActivo = TRUE THEN

	                        //  =============================================================================
	                        //  CUENTA FLOP
	                        IF "DB2045_TQ_AE".AE[#for_i].Aux_DuchaFlopActivo = TRUE THEN
	                            "DB2045_TQ_AE".AE[#for_i].Tiempos_DuchaTiempoFlop += 1;

	                        //  GESTION LOGICA Y SALIDAS FLIP

	                        //  GESTION LOGICA Y SALIDAS FLOP

	                    END_IF;
	                    
	                END_IF;
	                
	                //  =============================================================================
	                //  GESTION LOGICA SALIDA DUCHA
	                "DB2045_TQ_AE".AE[#for_i].Salida_ValvulaDucha :=
	                
	                (
	                NOT "DB2045_TQ_AE".AE[#for_i].Estado_AutoMan
	                AND
	                (
	                "DB2045_TQ_AE".AE[#for_i].Orden_Produccion
	                OR
	                ("DB2045_TQ_AE".AE[#for_i].Orden_CIP AND "DB2045_TQ_AE".AE[#for_i].Aux_DuchaSalidaFlip)

	                //  GESTION LOGICA SALIDA VENTEO

	                (
	                (NOT "DB2045_TQ_AE".AE[#for_i].Estado_AutoMan AND "DB2045_TQ_AE".AE[#for_i].Orden_Produccion)
	                OR
	                ("DB2045_TQ_AE".AE[#for_i].Estado_AutoMan AND "DB2045_TQ_AE".AE[#for_i].Orden_Manual)

	                NOT "DB2045_TQ_AE".AE[#for_i].Alarma_PresionAlta AND NOT "DB2045_TQ_AE".AE[#for_i].Alarma_PresionBaja;
	                
	            END_REGION
	            
	            
	            REGION VALVULA_ESTERIL
	                
	                //  =============================================================================
	                //  GESTION LOGICA SALIDA ESTERIL
	                IF (NOT "DB2045_TQ_AE".AE[#for_i].Estado_AutoMan AND "DB2045_TQ_AE".AE[#for_i].Orden_Produccion)
	                    OR
	                    ("DB2045_TQ_AE".AE[#for_i].Estado_AutoMan AND "DB2045_TQ_AE".AE[#for_i].Orden_Manual)

	                    IF NOT "DB2045_TQ_AE".AE[#for_i].Alarma_Configuracion AND NOT "DB2045_TQ_AE".AE[#for_i].Alarma_PresionBaja THEN

	                        END_IF;

	                            "DB2045_TQ_AE".AE[#for_i].Salida_ValvulaEsteril := FALSE;
	                        END_IF;
	                        
	                    ELSE
	                        "DB2045_TQ_AE".AE[#for_i].Salida_ValvulaEsteril := FALSE;
	                    END_IF;
	                ELSE
	                    "DB2045_TQ_AE".AE[#for_i].Salida_ValvulaEsteril := FALSE;
	                END_IF;
	                
	            END_REGION
	            
	            
	            REGION TRAZABILIDAD_MANUALIZACION
	                
	                IF "DB2045_TQ_AE".AE[#for_i].Estado_AutoMan AND NOT "DB2045_TQ_AE".AE[#for_i].Aux_oldAutoMan THEN
	                    "FC8_ZC_TRAZA_REGISTRO"(FechaHora := "DB1_SYS".FechaHoraActual,
	                                            Registrar := 1,
	                                            Categoria := "TRZ_CAT_3_DISP",
	                                            User := "DB1_SYS".Usuario.Actual,
	                                            Codigo1 := "TRZ_DISP_TIPO_TQ_AE",
	                                            Codigo2 := "DB2045_TQ_AE".AE[#for_i].Hmi_Index,
	                                            Codigo3 := "TRZ_DISP_ACC_MAN",
	                                            Texto := '');
	                END_IF;
	                
	                IF NOT "DB2045_TQ_AE".AE[#for_i].Estado_AutoMan AND "DB2045_TQ_AE".AE[#for_i].Aux_oldAutoMan THEN
	                    "FC8_ZC_TRAZA_REGISTRO"(FechaHora := "DB1_SYS".FechaHoraActual,
	                                            Registrar := 1,
	                                            Categoria := "TRZ_CAT_3_DISP",
	                                            User := "DB1_SYS".Usuario.Actual,
	                                            Codigo1 := "TRZ_DISP_TIPO_TQ_AE",
	                                            Codigo2 := "DB2045_TQ_AE".AE[#for_i].Hmi_Index,
	                                            Codigo3 := "TRZ_DISP_ACC_AUTO",
	                                            Texto := '');
	                END_IF;
	                
	                IF "DB2045_TQ_AE".AE[#for_i].Estado_AutoMan AND "DB2045_TQ_AE".AE[#for_i].Orden_Manual AND NOT "DB2045_TQ_AE".AE[#for_i].Aux_oldOrdenManual THEN
	                    "FC8_ZC_TRAZA_REGISTRO"(FechaHora := "DB1_SYS".FechaHoraActual,
	                                            Registrar := 1,
	                                            Categoria := "TRZ_CAT_3_DISP",
	                                            User := "DB1_SYS".Usuario.Actual,
	                                            Codigo1 := "TRZ_DISP_TIPO_TQ_AE",
	                                            Codigo2 := "DB2045_TQ_AE".AE[#for_i].Hmi_Index,
	                                            Codigo3 := "TRZ_DISP_ACC_MAN_ON",
	                                            Texto := '');
	                END_IF;
	                
	                IF "DB2045_TQ_AE".AE[#for_i].Estado_AutoMan AND NOT "DB2045_TQ_AE".AE[#for_i].Orden_Manual AND "DB2045_TQ_AE".AE[#for_i].Aux_oldOrdenManual THEN
	                    "FC8_ZC_TRAZA_REGISTRO"(FechaHora := "DB1_SYS".FechaHoraActual,
	                                            Registrar := 1,
	                                            Categoria := "TRZ_CAT_3_DISP",
	                                            User := "DB1_SYS".Usuario.Actual,
	                                            Codigo1 := "TRZ_DISP_TIPO_TQ_AE",
	                                            Codigo2 := "DB2045_TQ_AE".AE[#for_i].Hmi_Index,
	                                            Codigo3 := "TRZ_DISP_ACC_MAN_OFF",
	                                            Texto := '');
	                END_IF;
	                
	            END_REGION
	            
	            
	            // =============================================================================
	            //  Estado anterior manual/automatico
	            "DB2045_TQ_AE".AE[#for_i].Aux_oldOrdenManual := "DB2045_TQ_AE".AE[#for_i].Orden_Manual;
	            "DB2045_TQ_AE".AE[#for_i].Aux_oldAutoMan := "DB2045_TQ_AE".AE[#for_i].Estado_AutoMan;
	            
	        ELSE
	            
	            // =============================================================================
	            //  INICIALIZACIÓN DE SALIDAS
	            "DB2045_TQ_AE".AE[#for_i].Estado_AutoMan :=
	            "DB2045_TQ_AE".AE[#for_i].Orden_Manual :=
	            "DB2045_TQ_AE".AE[#for_i].Orden_Produccion :=
	            "DB2045_TQ_AE".AE[#for_i].Orden_CIP :=
	            "DB2045_TQ_AE".AE[#for_i].Orden_Vaciado :=
	            "DB2045_TQ_AE".AE[#for_i].Alarma_Configuracion :=
	            "DB2045_TQ_AE".AE[#for_i].Alarma_PresionAlta :=
	            "DB2045_TQ_AE".AE[#for_i].Alarma_PresionBaja :=
	            "DB2045_TQ_AE".AE[#for_i].Salida_ValvulaDucha :=
	            "DB2045_TQ_AE".AE[#for_i].Salida_ValvulaEsteril :=
	            "DB2045_TQ_AE".AE[#for_i].Salida_ValvulaVenteo := FALSE;
	            
	        END_IF;
