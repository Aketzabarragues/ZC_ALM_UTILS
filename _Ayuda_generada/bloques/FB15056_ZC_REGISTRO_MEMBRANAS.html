REGION ORDEN REGISTRO
    #Pulsador.Permiso := (#s_TecnicoNuevoRegistro <> '');
    "FC15026_ZC_PULSADOR"(Pulso1Seg := #Pulso1Seg,
                          SP_Tiempo := 2,
                          Categoria := "TRZ_CAT_4_PUL",
                          Codigo1 := "TRZ_PUL_MEM_CARE",
                          Codigo2 := 0,
                          Codigo3 := 0,
                          Usuario := #Usuario,
                          FechaHoraActual := #FechaHora,
                          Pulsador := #Pulsador);

    //  Orden pulsador
    IF #Pulsador.Plc THEN

        #Registros[#s_Indice].FechaHora := "FC15005_ZC_DTL_TO_ZC_FECHA"( #FechaHora);
        #Registros[#s_Indice].Tecnico := #s_TecnicoNuevoRegistro;
        #Registros[#s_Indice].TiempoMembrana := #Tiempo.Hora;

        //  Reset valor técnico y orden reset tiempo
        #s_TecnicoNuevoRegistro := '';

        #t_OrdenReset := TRUE;

        //  Incrementamos indice y si es mayor que el número máximo de registro lo ponemos en la posición inicial
        #s_Indice += 1;
        IF #s_Indice > #N_REGISTROS THEN
            #s_Indice := 1;
        END_IF;
    END_IF;
END_REGION
